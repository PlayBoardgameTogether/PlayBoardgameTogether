const COLORS = ['red', 'yellow', 'green', 'blue'];
const COPIES = {
  '0': 2, '1': 2, '2': 2, '3': 2, '4': 2, '5': 2, '6': 2, '7': 2, '8': 2, '9': 2,
  skip: 3, reverse: 3, draw2: 3, draw4: 2, skipAll: 2, discardAll: 3
};
const WILDS = { reverse4: 8, draw6: 4, draw10: 4, roulette: 8 };
const PENALTIES = { draw2: 2, draw4: 4, reverse4: 4, draw6: 6, draw10: 10 };
function deck() {
  const cards = [];
  for (const color of [...COLORS, 'wild']) {
    for (const [value, count] of Object.entries(color === 'wild' ? WILDS : COPIES)) {
      for (let copy = 0; copy < count; copy++) cards.push({ id: `${color}.${value}.${copy}`, color, value });
    }
  }
  return cards;
}
function shuffle(cards, rng) {
  for (let i = cards.length - 1; i > 0; i--) {
    const j = rng.nextInt(0, i + 1);
    [cards[i], cards[j]] = [cards[j], cards[i]];
  }
  return cards;
}
function active(state) {
  return state.seats.map((seat, index) => index).filter(index => !state.eliminated.includes(state.seats[index]));
}
function next(state, from = state.turn, steps = 1) {
  let index = from;
  for (let step = 0; step < steps; step++) {
    do { index = (index + state.direction + state.seats.length) % state.seats.length; }
    while (state.eliminated.includes(state.seats[index]));
  }
  return index;
}
function take(state, index, count, rng, until = null) {
  const cards = [];
  for (let i = 0; i < count; i++) {
    if (!state.deck.length) {
      const top = state.discard.pop();
      state.deck = shuffle([...state.discard, ...state.retired], rng);
      state.discard = [top];
      state.retired = [];
    }
    if (!state.deck.length) break;
    const card = state.deck.pop();
    state.hands[index].push(card);
    cards.push(card);
    if (state.hands[index].length >= 25) {
      state.eliminated.push(state.seats[index]);
      state.retired.push(...state.hands[index]);
      state.hands[index] = [];
      if (state.unoSeat === state.seats[index]) state.unoSeat = null;
      break;
    }
    if (until && until(card)) break;
  }
  return cards;
}
function matches(state, card) {
  return card.color === 'wild' || card.color === state.color || card.value === state.discard[state.discard.length - 1].value;
}
function canPlay(state, card) {
  if (state.drawn !== null) return card.id === state.drawn;
  if (state.pendingDraw) return (PENALTIES[card.value] || 0) >= state.lastDraw;
  return matches(state, card);
}
function actions(state, index) {
  if (index < 0 || state.winners.length || state.eliminated.includes(state.seats[index])) return [];
  const result = [];
  if (state.unoSeat !== null) {
    result.push(state.unoSeat === state.seats[index]
      ? { type: 'callUno' } : { type: 'catchUno', target: state.unoSeat });
  }
  if (index !== state.turn) return result;
  if (state.phase === 'swap') {
    return [...result, ...active(state).filter(other => other !== index).map(other => ({ type: 'swap', target: state.seats[other] }))];
  }
  if (state.phase === 'roulette') return [...result, ...COLORS.map(color => ({ type: 'roulette', color }))];
  const hand = state.hands[index];
  for (const card of hand) {
    if (!canPlay(state, card)) continue;
    // Roulette's recipient chooses the color in the following persistent phase.
    for (const color of card.color === 'wild' && card.value !== 'roulette' ? COLORS : [null]) {
      result.push({ type: 'play', cardId: card.id, color, uno: true });
    }
  }
  if (state.pendingDraw || (!state.drawn && !hand.some(card => canPlay(state, card)))) result.push({ type: 'draw' });
  return result;
}
function settle(state) {
  const remaining = active(state);
  if (remaining.length === 1) state.winners = [state.seats[remaining[0]]];
  if (!state.winners.length && (state.turns >= 2000 || state.stalled >= remaining.length)) {
    const minimum = Math.min(...remaining.map(index => state.hands[index].length));
    state.winners = remaining.filter(index => state.hands[index].length === minimum).map(index => state.seats[index]);
    state.last = '本局达到回合上限或无牌可摸，剩余玩家按最少手牌结算';
  }
  if (state.winners.length) {
    state.phase = 'play';
    state.drawn = null;
    state.unoSeat = null;
    state.pendingDraw = 0;
    state.lastDraw = 0;
  }
}
function endTurn(state, from = state.turn, steps = 1) {
  state.phase = 'play';
  state.drawn = null;
  state.turn = next(state, from, steps);
  state.turns++;
}
